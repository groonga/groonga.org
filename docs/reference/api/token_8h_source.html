26/*
27 * grn_tokenize_mode describes propose for tokenization.
28 *
29 * `GRN_TOKENIZE_GET`: Tokenize for search.
30 *
31 * `GRN_TOKENIZE_ADD`: Tokenize for adding token to index.
32 *
33 * `GRN_TOKENIZE_DELETE`: Tokenize for deleting token from index.
34 *
35 * @since 4.0.8

62/*
63 * grn_token_status is a flag set for tokenizer status codes.
64 * If a document or query contains no tokens, push an empty string with
65 * GRN_TOKEN_LAST as a token.
66 *
67 * @since 4.0.8
68 */
69typedef uint32_t grn_token_status;
70
71/*
72 * GRN_TOKEN_CONTINUE means that the next token is not the last one.
73 *
74 * @since 4.0.8
75 */
76#define GRN_TOKEN_CONTINUE (0)
77/*
78 * GRN_TOKEN_LAST means that the next token is the last one.
79 *
80 * @since 4.0.8
81 */
82#define GRN_TOKEN_LAST (0x01L << 0)
83/*
84 * GRN_TOKEN_OVERLAP means that ...
85 *
86 * @since 4.0.8
87 */
88#define GRN_TOKEN_OVERLAP (0x01L << 1)
89/*
90 * GRN_TOKEN_UNMATURED means that ...
91 *
92 * @since 4.0.8
93 */
94#define GRN_TOKEN_UNMATURED (0x01L << 2)
95/*
96 * GRN_TOKEN_REACH_END means that ...
97 *
98 * @since 4.0.8
99 */
100#define GRN_TOKEN_REACH_END (0x01L << 3)
101/*
102 * GRN_TOKEN_SKIP means that the token is skipped
103 *
104 * @since 4.0.8
105 */
106#define GRN_TOKEN_SKIP (0x01L << 4)
107/*
108 * GRN_TOKEN_SKIP_WITH_POSITION means that the token and postion is skipped
109 *
110 * @since 4.0.8
111 */
112#define GRN_TOKEN_SKIP_WITH_POSITION (0x01L << 5)
113/*
114 * GRN_TOKEN_FORCE_PREFIX means that the token is used common prefix search
115 *
116 * @since 4.0.8
117 */
118#define GRN_TOKEN_FORCE_PREFIX (0x01L << 6)
119/*
120 * GRN_TOKEN_KEEP_ORIGINAL means that the original token of the token is
121 * also used. This is for token filter.
122 *
123 * @since 11.0.3
124 */
125#define GRN_TOKEN_KEEP_ORIGINAL (0x01L << 7)
